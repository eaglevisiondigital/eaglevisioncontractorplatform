async function ownedReaderFixture(
  f: Awaited<ReturnType<typeof setup>>,
  options: {
    deadlineMs?: number;
    cancelTransport?: 'pinned-driver' | 'unavailable-fixture';
    missingProbe?: boolean;
    missingRelease?: boolean;
    holdCleanup?: () => Promise<void>;
  } = {},
) {
  const present = async () => {
    await options.holdCleanup?.();
    if (options.missingProbe)
      throw Error('CONTROLLED MISSING BACKEND OBSERVATION');
    const r = await authDb.query<{ present: boolean }>(
      'select exists(select 1 from pg_stat_activity where datname=$1 and pid=$2 and backend_start::text=$3) present',
      [f.backend.database, f.backend.pid, f.backend.backend_start],
    );
    return r.rows[0].present;
  };
  const owner = disposableReaderOwner(f.raw, f.backend, present, {
    enabled: true,
    mode: 'owned-pg17-unix-socket-fixture',
    deadlineMs: options.deadlineMs,
    cancelTransport: options.cancelTransport,
  });
  const claims = (
    await f.raw.query("select current_setting('request.jwt.claims') claims")
  ).rows[0].claims;
  const bearer =
    Buffer.from('{}').toString('base64url') +
    '.' +
    Buffer.from(claims).toString('base64url') +
    '.controlled_native_fixture';
  // Separate original fixture reader is the trusted release-authority witness.
  // Its connection remains fixture-owned until the original afterEach closes it;
  // no awaited witness cleanup is inserted after the final authority observation.
  const witness = await setup();
  let releaseChecks = 0;
  const releaseCurrent = async (
    target: CommercialCustomerTarget,
    expected: CommercialCustomerResponse,
    signal: AbortSignal,
  ) => {
    releaseChecks++;
    if (options.missingRelease)
      throw Error('CONTROLLED MISSING RELEASE AUTHORITY');
    const fresh = await witness.load(target, signal);
    expect(fresh.invoice).toEqual(expected.invoice);
    expect(fresh.payment).toEqual(expected.payment);
    expect(fresh.link.status).toBe('DISABLED');
  };
  return {
    owner,
    first: owner.reader(f.auth, bearer, binding, releaseCurrent),
    alias: owner.reader(f.auth, bearer, binding, releaseCurrent),
    releaseChecks: () => releaseChecks,
  };
}
async function blockedOn(f: Awaited<ReturnType<typeof setup>>) {
  for (let i = 0; i < 200; i++) {
    const r = await authDb.query<{ w: string }>(
      'select wait_event_type w from pg_stat_activity where pid=$1',
      [f.backend.pid],
    );
    if (r.rows[0]?.w === 'Lock') return;
    await new Promise((resolve) => setTimeout(resolve, 5));
  }
  throw Error('Original reader SQL did not enter the observed lock wait');
}

async function ownsAbsent(f: Awaited<ReturnType<typeof setup>>) {
  return (
    (
      await authDb.query(
        'select pid from pg_stat_activity where datname=$1 and pid=$2 and backend_start::text=$3',
        [f.backend.database, f.backend.pid, f.backend.backend_start],
      )
    ).rows.length === 0

}

test('physical custody qualification: observed abort cancels actual SQL57014 and discards the exact backend while the blocker remains', async () => {
  const f = await setup(),
    before = await invoiceFingerprint(),
    p = await ownedReaderFixture(f),
    blocker = await db.fork!(),
    controller = new AbortController();
  expect(
    (await f.raw.query('show statement_timeout')).rows[0].statement_timeout,
  ).toBe('0');
  expect((await f.raw.query('show lock_timeout')).rows[0].lock_timeout).toBe(
    '0',
  );
  let pending: Promise<string> | undefined;
  try {
    await blocker.exec(
      'begin;lock table private.commercial_invoice_drafts in access exclusive mode',
    );
    pending = p.first(f.viewer, controller.signal).then(
      () => 'UNEXPECTED_DATA',
      () => 'UNKNOWN',
    );
    await blockedOn(f);
    controller.abort();
    expect(await pending).toBe('UNKNOWN');
    expect(p.owner.snapshot()).toMatchObject({
      cancelRequested: true,
      sqlCancellation: true,
      socketEnded: true,
      backendAbsent: true,
      cleanupUncertain: false,
    });
    expect(p.owner.snapshot().sqlStates).toContain('57014');
    expect(await ownsAbsent(f)).toBe(true);
    await expect(f.raw.query('select 1')).rejects.toThrow();
    expect((await blocker.query('select 1 v')).rows[0].v).toBe(1);
  } finally {
    await blocker.exec('rollback');
    await pending;
    await blocker.close();
  }
  expect(await invoiceFingerprint()).toEqual(before);
});

function cleanupBarrier() {
  let entered!: () => void, resume!: () => void;
  const began = new Promise<void>((resolve) => {
    entered = resolve;
  });
  const released = new Promise<void>((resolve) => {
    resume = resolve;
  });
  return {
    began,
    resume,
    hold: async () => {
      entered();
      await released;
    },
  };
}
test('physical custody qualification: customer session revoked during awaited cleanup withholds a completed projection', async () => {
  const f = await setup(),
    barrier = cleanupBarrier(),
    p = await ownedReaderFixture(f, { holdCleanup: barrier.hold });
  const pending = p.first(f.viewer, sig()).then(
    () => 'UNEXPECTED_DATA',
    () => 'UNKNOWN',
  );
  try {
    await barrier.began;
    expect(p.owner.snapshot().completedRead).toBe(true);
    await authDb.query(
      "update auth.sessions set not_after=clock_timestamp()-interval '1 second' where id=$1",
      [f.c.customerSession],
    );
  } finally {
    barrier.resume();
  }
  expect(await pending).toBe('UNKNOWN');
  expect(p.releaseChecks()).toBe(1);
  expect(p.owner.snapshot()).toMatchObject({
    socketEnded: true,
    backendAbsent: true,
    releaseVerified: false,
    withheld: true,
  });
});
test('physical custody qualification: target changes during awaited cleanup cannot release the prior projection', async () => {
  const f = await setup(),

    socketEnded: true,

    releaseVerified: false,

  });
});
