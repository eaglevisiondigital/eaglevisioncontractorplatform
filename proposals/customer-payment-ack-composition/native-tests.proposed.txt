async function original(c = companies[0]) {
  const result = await observer<{
    invoice: Record<string, unknown>;
    version: EstimateVersion;
    source_current: boolean;
  }>(
    `select private.commercial_issue_projection(h.organization_id,h.id) invoice,to_jsonb(v) version,
  (p.status='issued' and p.current_version_id=v.id and v.status='issued' and v.customer_id=h.customer_id and v.location_id is not distinct from h.location_id and
    encode(sha256(convert_to(jsonb_build_object('version',v.id,'customer',v.customer_id,'location',v.location_id,'payload',v.payload,'calculation',v.calculation,'tax_policy',v.tax_policy_snapshot)::text,'UTF8')),'hex')=h.source_hash and not exists(select 1 from public.estimate_versions n where n.organization_id=v.organization_id and n.estimate_id=v.estimate_id and n.status='issued' and n.version_number>v.version_number)) source_current
  from private.commercial_invoice_drafts h join public.estimate_versions v on v.organization_id=h.organization_id and v.id=h.source_version_id join public.estimates p on p.organization_id=v.organization_id and p.id=v.estimate_id where h.organization_id=$1 and h.id=$2`,
    [c.org, draft.invoiceId],
  );
  if (result.rows.length !== 1 || result.rows[0].source_current !== true)
    throw Error('Fixture original source unavailable');
  return result.rows[0];
}

async function payment() {
  const f = await customer(),
    c = f.c;
  const merchant = {
    organizationId: c.org,
    merchantConfigId: merchantId,
    provider: 'nmi' as const,
    mode: 'sandbox' as const,
    approvalReference: 'FICTIONAL ONLY',
    approvedUntil: new Date(Date.now() + 3600000).toISOString(),
  };
  const intent = {
    organizationId: c.org,
    merchantConfigId: merchantId,
    invoiceId: draft.invoiceId,
    authorizationId,
    idempotencyKey: key,
    amountMinor: amount,
    currency: 'USD' as const,
    expiresAt: new Date(Date.now() + 120000).toISOString(),
  };
  let calls = 0;
  const connection = {
    async query<T extends Record<string, unknown>>(
      sql: string,
      args?: unknown[],
    ) {
      calls++;
      return { rows: (await observer<T>(sql, args)).rows };
    },
  };
  const input = { merchant, intent, requestDigest: digest };
  return {
    ...f,
    input,
    connection,
    calls: () => calls,
    load: () =>
      customerCaptureLinkLifecycle(
        f.auth,
        f.bearer,
        f.port,
        connection,
        input,
        { enabled: true, mode: 'disposable-proposal' },
      )(f.viewer, new AbortController().signal),
  };
}

  await finish('succeeded');
  const f = await payment();
  let original: OriginalInvoicePayment = {
    organizationId: f.c.org,
    merchantConfigId: merchantId,
    provider: 'nmi',
    mode: 'sandbox',
    invoiceId: draft.invoiceId,
    customerId: f.c.customer,
    sourceVersionId: f.c.version,
    sourceHash: draft.sourceHash,
    revision: draft.revision,
    issueCommandId: command,
    attemptKey: key,
    authorizationId,
    requestDigest: digest,
    kind: 'capture',
    captureKey: null,
  };
  if (kind === 'refund') {
    await observer(
      'insert into private.sandbox_gateway_settlements(organization_id,capture_key,evidence_sha256) values($1,$2,$3)',
      [f.c.org, key, 'a'.repeat(64)],
    );
    const refundAuth = randomUUID(),
      refundKey = randomUUID(),
      refundDigest = 'e'.repeat(64);
    await observer(
      `insert into private.sandbox_gateway_authorizations(organization_id,id,merchant_id,customer_id,invoice_id,provider,kind,amount_minor,currency,request_digest,evidence_reference,expires_at) values($1,$2,$3,$4,$5,'nmi','refund',100,'USD',$6,'FICTIONAL SEPARATE REFUND AUTHORITY',clock_timestamp()+interval '1 hour')`,
      [
        f.c.org,
        refundAuth,
        merchantId,
        f.c.customer,
        draft.invoiceId,
        refundDigest,
      ],
    );
    const store = new PostgresSandboxIntentStore(
      {
        async transaction(operation) {
          await observer('begin');
          try {
            const value = await operation({
              query: async <T extends Record<string, unknown>>(
                sql: string,
                p?: unknown[],
              ) => ({ rows: (await observer<T>(sql, p)).rows }),
            });
            await observer('commit');
            return value;
          } catch (error) {
            await observer('rollback');
            throw error;
          }
        },
      },
      f.c.org,
      merchantId,
    );
    const intent = {
      ...f.input.intent,
      authorizationId: refundAuth,
      idempotencyKey: refundKey,
      amountMinor: 100,
      captureKey: key,
      providerTransactionId: 'FICTIONAL_TRANSACTION',
    };
    expect(
      (await store.reserveRefund(f.input.merchant, intent, refundDigest))
        .status,
    ).toBe('reserved');
    await store.finishRefund(f.input.merchant, intent, {
      status: refundStatus,
      provider: 'nmi',
      mode: 'sandbox',
      transactionId: refundStatus === 'succeeded' ? 'FICTIONAL_REFUND' : null,
      amountMinor: 100,
      currency: 'USD',
    });
    original = {
      ...original,
      kind: 'refund',
      attemptKey: refundKey,
      authorizationId: refundAuth,
      requestDigest: refundDigest,
      captureKey: key,
    };
  }
  let reads = 0;
  const receiptConnection = {
    async query<T extends Record<string, unknown>>(
      sql: string,
      args?: unknown[],
    ) {
      reads++;
      return { rows: (await receiptDb.query<T>(sql, args)).rows };
    },
  };
  const run = () =>
    originalCustomerPaymentAck(
      f.auth,
      f.bearer,
      f.port,
      f.connection,
      receiptConnection,
      f.input,
      original,
      { enabled: true, mode: 'disposable-proposal' },
    )(f.viewer, new AbortController().signal);
  return { ...f, original, receiptConnection, run, reads: () => reads };
}

test('lost READ acknowledgment recovers the same original ACK by reads only', async () => {
  const f = await originalAck('refund'),
    before = await unchangedState(),
    query = f.receiptConnection.query;

  f.receiptConnection.query = async <T extends Record<string, unknown>>(
    sql: string,
    args?: unknown[],
  ) => {

      lose = false;

  expect(await unchangedState()).toEqual(before);
});
