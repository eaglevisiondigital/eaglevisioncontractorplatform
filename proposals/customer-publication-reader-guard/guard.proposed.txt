const executingBytes = readFileSync(sourceModule);
const rootIdentity = lstatSync(sourceRoot);
function currentModule() {
  const info = lstatSync(sourceRoot);
  if (
    resolve(process.cwd()) !== sourceRoot ||
    realpathSync(sourceRoot) !== sourceRoot ||
    !info.isDirectory() ||
    info.isSymbolicLink() ||
    info.dev !== rootIdentity.dev ||
    info.ino !== rootIdentity.ino
  )
    throw Error('Exact publication admission source required');
  let ancestor = sourceModule;
  while (true) {
    if (lstatSync(ancestor).isSymbolicLink())
      throw Error('Exact publication admission source required');
    const parent = dirname(ancestor);
    if (parent === ancestor) break;
    ancestor = parent;
  }
  const own = lstatSync(sourceModule);
  if (
    !own.isFile() ||
    (own.mode & 0o777) !== 0o644 ||
    !executingBytes.equals(readFileSync(sourceModule))
  )
    throw Error('Exact publication admission source required');
}
export function assertPublicationSources({
  read = readFileSync,
  stat = lstatSync,
}: { read?: typeof readFileSync; stat?: typeof lstatSync } = {}) {
  const checked = () => {
    currentModule();
    for (const [path, hash] of Object.entries(pins)) {
      const parts = path.split('/');
      for (let i = 1; i <= parts.length; i++) {
        const s = stat(parts.slice(0, i).join('/'));
        if (
          s.isSymbolicLink() ||
          (i < parts.length
            ? !s.isDirectory()
            : !s.isFile() || (s.mode & 0o777) !== 0o644)
        )
          throw Error('Exact publication admission source required');
      }
      if (
        createHash('sha256')
          .update(Buffer.from(read(path)))
          .digest('hex') !== hash
      )
        throw Error('Exact publication admission source required');
    }
  };
  checked();
  checked();
  currentModule();
}
